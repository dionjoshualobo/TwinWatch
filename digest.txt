rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null;
    }

    match /users/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }

    match /groups/{groupId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.auth.uid in request.resource.data.memberUids;
      allow update: if signedIn() && (
        request.auth.uid in resource.data.memberUids ||
        (
          request.auth.uid in request.resource.data.memberUids &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['memberUids'])
        )
      );
      allow delete: if signedIn() && request.auth.uid in resource.data.memberUids;

      match /members/{memberUid} {
        allow read: if signedIn();
        allow create: if signedIn() && request.auth.uid in request.resource.data.memberUids;
        allow update, delete: if signedIn() && request.auth.uid in resource.data.memberUids;
      }

      match /shows/{showId} {
        allow read: if signedIn();
        allow create: if signedIn() && request.auth.uid in request.resource.data.memberUids;
        allow update, delete: if signedIn() && request.auth.uid in resource.data.memberUids;
      }

      match /watched/{memberUid} {
        allow read: if signedIn();
        allow delete: if signedIn() && request.auth.uid == memberUid;

        match /episodes/{episodeId} {
          allow read: if signedIn();
          allow create, update: if signedIn() && request.auth.uid == memberUid &&
            request.auth.uid in request.resource.data.memberUids;
          allow delete: if signedIn() && (
            request.auth.uid == memberUid ||
            request.auth.uid in resource.data.memberUids
          );
        }
      }

      match /notes/{noteId} {
        allow read: if signedIn() && (resource.data.isPublic == true || resource.data.uid == request.auth.uid);
        allow create: if signedIn() &&
          request.auth.uid == request.resource.data.uid &&
          request.auth.uid in request.resource.data.memberUids;
        allow update: if signedIn() && resource.data.uid == request.auth.uid;
        allow delete: if signedIn() && (
          resource.data.uid == request.auth.uid ||
          request.auth.uid in resource.data.memberUids
        );
      }

      match /events/{eventId} {
        allow read: if signedIn();
        allow create: if signedIn() &&
          request.auth.uid == request.resource.data.uid &&
          request.auth.uid in request.resource.data.memberUids;
        allow update: if false;
        allow delete: if signedIn() && request.auth.uid in resource.data.memberUids;
      }
    }
  }
}

internal fun extractInviteCode(input: String): String {
    val trimmed = input.trim()
    if (trimmed.isBlank()) return ""
    val uri = runCatching { Uri.parse(trimmed) }.getOrNull()
    val raw = when {
        uri?.scheme == "watchtogether" && uri.host == "join" ->
            uri.getQueryParameter("code") ?: uri.lastPathSegment
        else -> trimmed
    } ?: trimmed
    return raw.filter { it.isLetterOrDigit() }.uppercase().take(6)
}

class MainActivity : ComponentActivity() {
    private val vm: MainViewModel by viewModels()

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        readIntent(intent)
        setContent {
            AppTheme(darkTheme = vm.darkTheme) {
                Surface(modifier = Modifier.fillMaxSize(), color = MaterialTheme.colorScheme.background) {
                    AppRoot(vm, intent.getStringExtra("groupId"))
                }
            }
        }
    }

    override fun onNewIntent(intent: Intent) {
        super.onNewIntent(intent)
        readIntent(intent)
    }

    private fun readIntent(intent: Intent?) {
        val uri = intent?.data ?: return
        if (uri.scheme == "watchtogether" && uri.host == "join") {
            val code = extractInviteCode(uri.toString())
            if (code.length >= 4) vm.submitJoinCode(code)
        }
    }
}

@Composable
fun AppRoot(vm: MainViewModel, startGroupId: String?) {
    val stack = remember { mutableStateListOf<Screen>(Screen.Splash) }
    val current = stack.last()

    BackHandler(enabled = stack.size > 1) { stack.removeAt(stack.size - 1) }

    val context = androidx.compose.ui.platform.LocalContext.current
    val notificationPermission = rememberLauncherForActivityResult(ActivityResultContracts.RequestPermission()) { }
    LaunchedEffect(vm.uid) {
        if (vm.uid != null && Build.VERSION.SDK_INT >= 33 &&
            ContextCompat.checkSelfPermission(context, Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) {
            notificationPermission.launch(Manifest.permission.POST_NOTIFICATIONS)
        }
    }

    fun go(screen: Screen) {
        stack.add(screen)
    }

    fun replace(screen: Screen) {
        stack.clear()
        stack.add(screen)
    }

    when (current) {
        Screen.Splash -> SplashScreen(vm, startGroupId, ::replace, ::go)
        Screen.Onboarding -> OnboardingScreen(vm) { replace(Screen.Groups) }
        Screen.Groups -> GroupsScreen(vm, ::go)
        Screen.Join -> JoinScreen(vm, ::replace, { stack.removeAt(stack.size - 1) })
        Screen.Settings -> SettingsScreen(vm) { stack.removeAt(stack.size - 1) }
        is Screen.GroupDetail -> GroupDetailScreen(vm, current.group, ::go)
        is Screen.MemberDetail -> MemberDetailScreen(vm, current.group, current.member, ::go)
        is Screen.ShowDetail -> ShowDetailScreen(vm, current.group, current.show, ::go)
        is Screen.GroupProgress -> GroupProgressScreen(vm, current.group, current.show, ::go)
        is Screen.AddShow -> AddShowScreen(vm, current.group) { stack.removeAt(stack.size - 1) }
    }

    vm.message?.let { msg ->
        AlertDialog(
            onDismissRequest = { vm.clearMessage() },
            containerColor = Brand.Surface,
            shape = RoundedCornerShape(24.dp),
            title = { Text("Something went wrong", color = Brand.Ink, fontWeight = FontWeight.Bold) },
            text = { Text(msg, color = Brand.Muted) },
            confirmButton = {
                PunchButton("OK", { vm.clearMessage() }, modifier = Modifier.width(110.dp), color = Brand.Coral)
            }
        )
    }
}

@Composable
fun SplashScreen(
    vm: MainViewModel,
    startGroupId: String?,
    replace: (Screen) -> Unit,
    go: (Screen) -> Unit
) {
    val uid = vm.uid
    LaunchedEffect(vm.ready, uid, vm.myName, vm.pendingJoinCode, startGroupId) {
        if (!vm.ready || uid == null) return@LaunchedEffect
        when {
            vm.myName.isBlank() -> replace(Screen.Onboarding)
            vm.pendingJoinCode != null -> replace(Screen.Join)
            startGroupId != null -> {
                try {
                    val group = vm.repo.group(startGroupId)
                    if (group != null) replace(Screen.GroupDetail(group)) else replace(Screen.Groups)
                } catch (e: Exception) {
                    if (!e.isCancellation()) vm.message = e.localizedMessage ?: "Could not open circle"
                }
            }
            else -> replace(Screen.Groups)
        }
    }
    DecorBackground {
        Column(
            modifier = Modifier.fillMaxSize().padding(32.dp),
            verticalArrangement = Arrangement.Center,
            horizontalAlignment = Alignment.CenterHorizontally
        ) {
            AnimatedEnter {
                SoftCard(modifier = Modifier.fillMaxWidth()) {
                    Column(Modifier.padding(28.dp), horizontalAlignment = Alignment.CenterHorizontally) {
                        Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                            repeat(6) {
                                Box(
                                    Modifier.size(width = 22.dp, height = 9.dp)
                                        .clip(RoundedCornerShape(9.dp))
                                        .background(listOf(Brand.Blue, Brand.Amber, Brand.Mint, Brand.Coral)[it % 4])
                                )
                            }
                        }
                        Spacer(Modifier.height(28.dp))
                        Text("TWIN", style = MaterialTheme.typography.displayLarge, color = Brand.Ink)
                        Text("WATCH", style = MaterialTheme.typography.displayLarge, color = Brand.Blue)

        }
    }
}

private fun pressScale(pressed: Boolean, enabled: Boolean): Modifier {

@Composable
fun PunchButton(
    text: String,
    onClick: () -> Unit,
    modifier: Modifier = Modifier,
    color: Color = Color.Unspecified,
    contentColor: Color = Color.Unspecified,
    borderColor: Color = Color.Unspecified,
    enabled: Boolean = true,
    haptic: Boolean = true,
    leadingIcon: (@Composable () -> Unit)? = null
) {
    val palette = LocalWatchPalette.current
    val resolvedColor = if (color == Color.Unspecified) palette.blue else color
    val resolvedContentColor = if (contentColor == Color.Unspecified) palette.onPrimary else contentColor
    val resolvedBorderColor = if (borderColor == Color.Unspecified) palette.stroke else borderColor
    val interaction = remember { MutableInteractionSource() }
    val pressed by interaction.collectIsPressedAsState()
    val feedback = LocalHapticFeedback.current
    val radius by animateDpAsState(
        if (pressed && enabled) 26.dp else 18.dp,
        spring(dampingRatio = 0.7f, stiffness = 520f),
        label = "buttonShape"
    )
    val shift by animateDpAsState(
        if (pressed && enabled) 4.dp else 0.dp,
        spring(dampingRatio = 0.7f, stiffness = 520f),
        label = "buttonShift"
    )
    val shadowShift by animateDpAsState(
        if (pressed && enabled) 0.dp else 5.dp,
        spring(dampingRatio = 0.7f, stiffness = 520f),
        label = "buttonShadow"
    )
    Box(modifier.height(56.dp).then(pressScale(pressed, enabled))) {
        Box(
            Modifier
                .matchParentSize()
                .offset(y = shadowShift)
                .clip(RoundedCornerShape(radius))
                .background(palette.shadow.copy(alpha = if (enabled) 1f else 0.4f))
        )
        Surface(
            shape = RoundedCornerShape(radius),
            color = if (enabled) resolvedColor else palette.paperDim,
            contentColor = if (enabled) resolvedContentColor else palette.muted,
            border = BorderStroke(2.dp, if (enabled) resolvedBorderColor else palette.border),
            modifier = Modifier
                .fillMaxWidth()
                .fillMaxHeight()
                .offset(y = shift)
                .clip(RoundedCornerShape(radius))
                .clickable(
                    interactionSource = interaction,
                    indication = LocalIndication.current,
                    enabled = enabled
                ) {
                    if (haptic) feedback.performHapticFeedback(HapticFeedbackType.TextHandleMove)
                    onClick()
                }
        ) {
            Row(
                Modifier.padding(horizontal = 18.dp),
                verticalAlignment = Alignment.CenterVertically,
                horizontalArrangement = androidx.compose.foundation.layout.Arrangement.Center
            ) {
                if (leadingIcon != null) {
                    leadingIcon()
                    Spacer(Modifier.width(10.dp))
                }
                Text(
                    text,
                    style = MaterialTheme.typography.labelLarge,
                    maxLines = 1,
                    overflow = androidx.compose.ui.text.style.TextOverflow.Ellipsis
                )
            }
        }
    }
}

internal fun Throwable.isCancellation(): Boolean {
    var cause: Throwable? = this
    while (cause != null) {
        if (cause is CancellationException) return true
        cause = cause.cause
    }
    return false
}

class MainViewModel(app: Application) : AndroidViewModel(app) {
    val repo = Repository()
    private val prefs = app.getSharedPreferences("watch_together_prefs", Context.MODE_PRIVATE)

    var ready by mutableStateOf(false)
        private set
    var uid by mutableStateOf<String?>(null)
        private set
    var myName by mutableStateOf("")
        private set
    var busy by mutableStateOf(false)
        private set
    var message by mutableStateOf<String?>(null)
    var pendingJoinCode by mutableStateOf<String?>(null)
        private set
    var darkTheme by mutableStateOf(prefs.getBoolean("dark_theme", false))
        private set

    init {
        viewModelScope.launch {
            try {
                val id = repo.ensureSignedIn()
                val name = repo.profile(id)?.name ?: ""
                uid = id
                myName = name
                ready = true
                startActivityWatcher(id)
                ReminderScheduler.schedule(getApplication())
            } catch (e: Exception) {
                if (!e.isCancellation()) message = e.localizedMessage ?: "Could not sign in"
            }
        }
    }

    private fun startActivityWatcher(uid: String) {
        viewModelScope.launch {
            repo.activityEventsFlow(uid).collect { (groupId, event) ->
                if (event.uid != uid && event.body.isNotBlank()) {
                    showActivityNotification(getApplication(), event.body, groupId)
                }
            }
        }
    }

    fun submitJoinCode(code: String?) {
        pendingJoinCode = code
    }

    fun saveName(name: String, onDone: () -> Unit) {
        viewModelScope.launch {
            busy = true
            try {
                val id = uid ?: repo.ensureSignedIn()
                uid = id
                repo.updateName(id, name.trim())
                myName = name.trim()
                onDone()
            } catch (e: Exception) {
                if (!e.isCancellation()) message = e.localizedMessage ?: "Could not save name"
            } finally {
                busy = false
            }
        }
    }

    fun setDarkMode(enabled: Boolean) {
        darkTheme = enabled
        prefs.edit().putBoolean("dark_theme", enabled).apply()
    }

    fun createGroup(name: String, onDone: (Group) -> Unit) {
        viewModelScope.launch {
            busy = true
            try {
                val id = uid ?: repo.ensureSignedIn()
                val group = repo.createGroup(name, id, myName)
                onDone(group)
            } catch (e: Exception) {
                if (!e.isCancellation()) message = e.localizedMessage ?: "Could not create group"
            } finally {
                busy = false
            }
        }
    }

    fun joinGroup(code: String, onDone: (Group?) -> Unit) {
        viewModelScope.launch {
            busy = true
            try {
                val id = uid ?: repo.ensureSignedIn()
                val group = repo.joinGroup(code, id, myName)
                if (group != null) pendingJoinCode = null
                onDone(group)
            } catch (e: Exception) {
                if (!e.isCancellation()) message = e.localizedMessage ?: "Could not join group"
            } finally {
                busy = false
            }
        }
    }

    fun launchApp(block: suspend () -> Unit) = viewModelScope.launch {
        try {
            block()
        } catch (e: Exception) {
            if (!e.isCancellation()) message = e.localizedMessage ?: "Something went wrong"
        }
    }

    fun clearMessage() {
        message = null
    }
}

class WatchMessagingService : FirebaseMessagingService() {
    override fun onMessageReceived(message: RemoteMessage) {
        val author = message.data["authorUid"] ?: message.data["uid"] ?: ""
        if (author.isNotEmpty() && author == Repository().currentUid()) return

        val body = message.data["body"]
            ?: message.notification?.body
            ?: "Something new happened in your group"
        showActivityNotification(this, body, message.data["groupId"] ?: "")
    }
}

fun showActivityNotification(context: Context, body: String, groupId: String) {
    val manager = context.getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
    if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Group activity",
            NotificationManager.IMPORTANCE_HIGH
        ).apply { description = "Episodes and notes from your groups" }
        manager.createNotificationChannel(channel)
    }

    val intent = Intent(context, MainActivity::class.java).apply {
        flags = Intent.FLAG_ACTIVITY_SINGLE_TOP or Intent.FLAG_ACTIVITY_CLEAR_TOP
        if (groupId.isNotEmpty()) putExtra("groupId", groupId)
    }
    val pendingIntent = PendingIntent.getActivity(
        context,
        groupId.hashCode(),
        intent,
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )

    val notification = NotificationCompat.Builder(context, CHANNEL_ID)
        .setSmallIcon(R.drawable.ic_launcher_foreground)
        .setContentTitle("TwinWatch")
        .setContentText(body)
        .setStyle(NotificationCompat.BigTextStyle().bigText(body))
        .setAutoCancel(true)
        .setPriority(NotificationCompat.PRIORITY_HIGH)
        .setContentIntent(pendingIntent)
        .build()

    manager.notify((groupId + body).hashCode(), notification)

package com.dionlobo.watchtogether

import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build
import androidx.core.app.NotificationCompat
import androidx.work.ExistingWorkPolicy
import androidx.work.OneTimeWorkRequestBuilder
import androidx.work.WorkManager
import androidx.work.Worker
import androidx.work.WorkerParameters
import androidx.work.workDataOf
import java.util.Calendar
import java.util.concurrent.TimeUnit

object ReminderScheduler {
    private const val CHANNEL_ID = "watch_together_reminders"
    private const val KEY_SLOT = "slot"
    private const val KEY_TITLE = "title"
    private const val KEY_BODY = "body"

    fun schedule(context: Context) {
        scheduleSlot(context, "midday", 13, 0, "Watch check-in", "Did you watch anything today? Tap to log it.")
        scheduleSlot(context, "evening", 21, 0, "Tonight's recap", "Log the episodes you finished tonight.")
    }

    fun scheduleSlot(context: Context, slot: String, hour: Int, minute: Int, title: String, body: String) {
        val work = OneTimeWorkRequestBuilder<ReminderWorker>()
            .setInitialDelay(millisUntil(hour, minute), TimeUnit.MILLISECONDS)
            .setInputData(
                workDataOf(
                    KEY_SLOT to slot,
                    KEY_TITLE to title,
                    KEY_BODY to body
                )
            )
            .build()
        WorkManager.getInstance(context)
            .enqueueUniqueWork("watch_reminder_$slot", ExistingWorkPolicy.REPLACE, work)
    }

    private fun millisUntil(hour: Int, minute: Int): Long {
        val now = Calendar.getInstance()
        val due = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, hour)
            set(Calendar.MINUTE, minute)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
        }
        if (due.timeInMillis <= now.timeInMillis) due.add(Calendar.DATE, 1)
        return due.timeInMillis - now.timeInMillis
    }

    fun createChannel(context: Context) {
        if (Build.VERSION.SDK_INT < Build.VERSION_CODES.O) return
        val manager = context.getSystemService(NotificationManager::class.java) ?: return
        if (manager.getNotificationChannel(CHANNEL_ID) != null) return
        val channel = NotificationChannel(
            CHANNEL_ID,
            "Watch reminders",
            NotificationManager.IMPORTANCE_DEFAULT
        )
        channel.description = "Daily prompts to log watched episodes"
        manager.createNotificationChannel(channel)
    }

    fun show(context: Context, slot: String, title: String, body: String) {
        createChannel(context)
        val intent = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TOP
            putExtra("openLog", true)
        }
        val flags = PendingIntent.FLAG_UPDATE_CURRENT or
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) PendingIntent.FLAG_IMMUTABLE else 0
        val pending = PendingIntent.getActivity(context, slot.hashCode(), intent, flags)
        val notification = NotificationCompat.Builder(context, CHANNEL_ID)
            .setSmallIcon(R.drawable.ic_launcher_foreground)
            .setContentTitle(title)
            .setContentText(body)
            .setStyle(NotificationCompat.BigTextStyle().bigText(body))
            .setContentIntent(pending)
            .setAutoCancel(true)
            .build()
        val manager = context.getSystemService(NotificationManager::class.java) ?: return
        manager.notify(slot.hashCode(), notification)
    }
}

class ReminderWorker(context: Context, params: WorkerParameters) : Worker(context, params) {
    override fun doWork(): Result {
        val slot = inputData.getString("slot") ?: "midday"
        val title = inputData.getString("title") ?: "Watch check-in"
        val body = inputData.getString("body") ?: "Did you watch anything today? Tap to log it."
        ReminderScheduler.show(applicationContext, slot, title, body)
        val hour = if (slot == "evening") 21 else 13
        val minute = 0
        ReminderScheduler.scheduleSlot(applicationContext, slot, hour, minute, title, body)
        return Result.success()
    }
}

package com.dionlobo.watchtogether

import com.google.firebase.auth.FirebaseAuth
import com.google.firebase.firestore.CollectionReference
import com.google.firebase.firestore.FieldValue
import com.google.firebase.firestore.FirebaseFirestore
import com.google.firebase.firestore.ListenerRegistration
import com.google.firebase.firestore.Query
import com.google.firebase.messaging.FirebaseMessaging
import kotlinx.coroutines.channels.awaitClose
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.callbackFlow
import kotlinx.coroutines.tasks.await
import kotlin.random.Random

class Repository {
    private val auth = FirebaseAuth.getInstance()
    private val db = FirebaseFirestore.getInstance()
    private val messaging = FirebaseMessaging.getInstance()

    fun currentUid(): String? = auth.currentUser?.uid

    suspend fun ensureSignedIn(): String {
        auth.currentUser?.let { return it.uid }
        return auth.signInAnonymously().await().user!!.uid
    }

    suspend fun saveProfile(uid: String, name: String) {
        db.collection("users").document(uid)
            .set(mapOf("name" to name, "createdAt" to FieldValue.serverTimestamp()))
            .await()
    }

    suspend fun profile(uid: String): UserProfile? {
        val snap = db.collection("users").document(uid).get().await()
        return if (snap.exists()) snap.toUserProfile() else null
    }

    suspend fun updateName(uid: String, name: String) {
        val clean = name.trim()
        val groups = db.collection("groups").whereArrayContains("memberUids", uid).get().await()
        db.runBatch { batch ->
            batch.set(
                db.collection("users").document(uid),
                mapOf("name" to clean, "createdAt" to FieldValue.serverTimestamp())
            )
            groups.documents.forEach { group ->
                batch.update(
                    group.reference.collection("members").document(uid),
                    mapOf("name" to clean)
                )
            }
        }.await()
    }

    fun groupsFlow(uid: String): Flow<List<Group>> = callbackFlow {
        val listener = db.collection("groups").whereArrayContains("memberUids", uid)
            .addSnapshotListener { snap, error ->
                if (error != null || snap == null) return@addSnapshotListener
                trySend(snap.documents.map { it.toGroup() }.sortedBy { it.name })
            }
        awaitClose { listener.remove() }
    }

    fun membersFlow(groupId: String): Flow<List<Member>> = callbackFlow {
        val listener = db.collection("groups").document(groupId).collection("members")
            .addSnapshotListener { snap, error ->
                if (error != null || snap == null) return@addSnapshotListener
                trySend(snap.documents.map { it.toMember() }.sortedBy { it.name })
            }
        awaitClose { listener.remove() }
    }

    fun showsFlow(groupId: String): Flow<List<ShowEntry>> = callbackFlow {
        val listener = db.collection("groups").document(groupId).collection("shows")
            .orderBy("addedAt", Query.Direction.ASCENDING)
            .addSnapshotListener { snap, error ->
                if (error != null || snap == null) return@addSnapshotListener
                trySend(snap.documents.map { it.toShowEntry() })
            }
        awaitClose { listener.remove() }
    }

    fun watchedFlow(groupId: String, uid: String): Flow<List<WatchedEpisode>> = callbackFlow {
        val listener = db.collection("groups").document(groupId)
            .collection("watched").document(uid).collection("episodes")
            .addSnapshotListener { snap, error ->
                if (error != null || snap == null) return@addSnapshotListener
                trySend(snap.documents.map { it.toWatchedEpisode() })
            }
        awaitClose { listener.remove() }
    }

    fun memberWatchedFlow(groupId: String, memberUids: List<String>): Flow<Map<String, List<WatchedEpisode>>> = callbackFlow {
        val watched = memberUids.associateWith { emptyList<WatchedEpisode>() }.toMutableMap()
        val listeners = memberUids.map { uid ->
            db.collection("groups").document(groupId).collection("watched")
                .document(uid).collection("episodes")
                .addSnapshotListener { snap, error ->
                    if (error != null || snap == null) return@addSnapshotListener
                    watched[uid] = snap.documents.map { it.toWatchedEpisode() }
                        .sortedWith(compareBy({ it.season }, { it.number }))
                    trySend(watched.toMap())
                }
        }
        trySend(watched.toMap())
        awaitClose { listeners.forEach { it.remove() } }
    }

    fun notesFlow(groupId: String, uid: String): Flow<List<Note>> = callbackFlow {
        val base = db.collection("groups").document(groupId).collection("notes")
        val publicNotes = mutableMapOf<String, Note>()
        val myNotes = mutableMapOf<String, Note>()

        fun publish() {
            val all = (publicNotes.values + myNotes.values)
                .distinctBy { it.id }
                .sortedByDescending { it.createdAt }
            trySend(all)
        }

        fun read(snap: com.google.firebase.firestore.QuerySnapshot?, map: MutableMap<String, Note>) {
            if (snap == null) return
            map.clear()
            snap.documents.forEach { map[it.id] = it.toNote() }
            publish()
        }

        val l1 = base.whereEqualTo("isPublic", true)
            .addSnapshotListener { snap, _ -> read(snap, publicNotes) }
        val l2 = base.whereEqualTo("uid", uid)
            .addSnapshotListener { snap, _ -> read(snap, myNotes) }
        awaitClose { l1.remove(); l2.remove() }
    }

    fun memberPublicNotesFlow(groupId: String, memberUid: String): Flow<List<Note>> = callbackFlow {
        val listener = db.collection("groups").document(groupId).collection("notes")
            .whereEqualTo("isPublic", true)
            .addSnapshotListener { snap, _ ->
                val notes = snap?.documents.orEmpty().map { it.toNote() }.filter { it.uid == memberUid }
                trySend(notes.sortedByDescending { it.createdAt })
            }
        awaitClose { listener.remove() }
    }

    fun eventsFlow(groupId: String): Flow<List<GroupEvent>> = callbackFlow {
        val listener = db.collection("groups").document(groupId).collection("events")
            .orderBy("createdAt", Query.Direction.DESCENDING).limit(60)
            .addSnapshotListener { snap, error ->
                if (error != null || snap == null) return@addSnapshotListener
                trySend(snap.documents.map { it.toGroupEvent() })
            }
        awaitClose { listener.remove() }
    }

    fun activityEventsFlow(uid: String): Flow<Pair<String, GroupEvent>> = callbackFlow {
        val eventListeners = mutableListOf<ListenerRegistration>()
        val seen = mutableMapOf<String, MutableSet<String>>()
        val initialized = mutableSetOf<String>()

        val groupsListener = db.collection("groups").whereArrayContains("memberUids", uid)
            .addSnapshotListener { groupSnap, _ ->
                eventListeners.forEach { it.remove() }
                eventListeners.clear()

                groupSnap?.documents?.forEach { groupDoc ->
                    val groupId = groupDoc.id
                    val eventIds = seen.getOrPut(groupId) { mutableSetOf() }
                    val firstSnapshot = initialized.add(groupId)

                    val listener = db.collection("groups").document(groupId).collection("events")
                        .addSnapshotListener { snap, error ->
                            if (error != null || snap == null) return@addSnapshotListener
                            if (snap.metadata.hasPendingWrites()) return@addSnapshotListener

                            val fresh = if (firstSnapshot) {
                                snap.documents.forEach { eventIds.add(it.id) }
                                emptyList()
                            } else {
                                snap.documents.filter { !eventIds.contains(it.id) }
                            }
                            fresh.forEach { eventIds.add(it.id) }
                            fresh.forEach { trySend(groupId to it.toGroupEvent()) }
                        }
                    eventListeners.add(listener)
                }
            }

        awaitClose {
            groupsListener.remove()
            eventListeners.forEach { it.remove() }
        }
    }

    suspend fun group(groupId: String): Group? {
        val snap = db.collection("groups").document(groupId).get().await()
        return if (snap.exists()) snap.toGroup() else null
    }

    suspend fun groupByCode(code: String): Group? {
        val snap = db.collection("groups").whereEqualTo("inviteCode", code.trim().uppercase()).limit(1).get().await()
        return snap.documents.firstOrNull()?.toGroup()
    }

    private suspend fun memberIds(groupId: String): List<String> =
        group(groupId)?.memberUids ?: emptyList()

    private fun newCode(): String {
        val alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
        return (1..6).map { alphabet[Random.nextInt(alphabet.length)] }.joinToString("")
    }

    suspend fun createGroup(name: String, uid: String, myName: String): Group {
        val ref = db.collection("groups").document()
        var code = newCode()
        while (groupByCode(code) != null) code = newCode()
        val group = Group(ref.id, name.trim(), code, uid, System.currentTimeMillis(), listOf(uid))
        ref.set(
            mapOf(
                "name" to group.name,
                "inviteCode" to group.inviteCode,
                "createdBy" to uid,
                "createdAt" to group.createdAt,
                "memberUids" to group.memberUids
            )
        ).await()
        addMember(ref.id, uid, myName, group.memberUids)
        subscribe(ref.id)
        return group
    }

    suspend fun joinGroup(code: String, uid: String, myName: String): Group? {
        val group = groupByCode(code) ?: return null
        if (!group.memberUids.contains(uid)) {
            db.collection("groups").document(group.id)
                .update("memberUids", FieldValue.arrayUnion(uid)).await()
            addMember(group.id, uid, myName, group.memberUids + uid)
        }
        subscribe(group.id)
        return group
    }

    private suspend fun addMember(groupId: String, uid: String, name: String, memberUids: List<String>) {
        db.collection("groups").document(groupId).collection("members").document(uid)
            .set(
                mapOf(
                    "uid" to uid,
                    "name" to name,
                    "memberUids" to memberUids,
                    "joinedAt" to System.currentTimeMillis()
                )
            )
            .await()
    }

    suspend fun subscribe(groupId: String) {
        runCatching { messaging.subscribeToTopic("g_$groupId").await() }
    }

    suspend fun addShow(groupId: String, show: TvShow, uid: String) {
        val ref = db.collection("groups").document(groupId).collection("shows").document(show.id.toString())
        ref.set(
            mapOf(
                "name" to show.name,
                "imageUrl" to (show.image?.medium ?: show.image?.original ?: ""),
                "addedBy" to uid,
                "memberUids" to memberIds(groupId),
                "addedAt" to System.currentTimeMillis()
            )
        ).await()
    }

    suspend fun markEpisodes(
        groupId: String,
        uid: String,
        authorName: String,
        show: ShowEntry,
        episodes: List<TvEpisode>,
        summary: String
    ) {
        val watched = db.collection("groups").document(groupId)
            .collection("watched").document(uid).collection("episodes")
        val groupMemberUids = memberIds(groupId)
        db.runBatch { batch ->
            episodes.forEach { ep ->
                batch.set(
                    watched.document(ep.id.toString()),
                    mapOf(
                        "season" to ep.season,
                        "number" to ep.number,
                        "name" to ep.name,
                        "showId" to show.id,
                        "memberUids" to groupMemberUids,
                        "markedAt" to System.currentTimeMillis()
                    )
                )
            }
            batch.set(
                db.collection("groups").document(groupId).collection("events").document(),
                mapOf(
                    "uid" to uid,
                    "authorName" to authorName,
                    "showId" to show.id,
                    "memberUids" to groupMemberUids,
                    "body" to "$authorName $summary of ${show.name}",
                    "createdAt" to System.currentTimeMillis()
                )
            )
        }.await()
    }

    suspend fun unmarkEpisode(groupId: String, uid: String, episodeId: Int) {
        db.collection("groups").document(groupId).collection("watched")
            .document(uid).collection("episodes").document(episodeId.toString()).delete().await()
    }

    suspend fun resetShowProgress(groupId: String, uid: String, episodeIds: List<String>) {
        val episodes = db.collection("groups").document(groupId).collection("watched")
            .document(uid).collection("episodes")
        episodeIds.chunked(300).forEach { chunk ->
            db.runBatch { batch ->
                chunk.forEach { id -> batch.delete(episodes.document(id)) }
            }.await()
        }
    }

    suspend fun removeShow(groupId: String, showId: String) {
        db.collection("groups").document(groupId).collection("shows").document(showId).delete().await()
    }

    suspend fun deleteGroup(groupId: String) {
        val root = db.collection("groups").document(groupId)
        val watched = root.collection("watched")
        watched.get().await().documents.forEach { member ->
            deleteCollection(member.reference.collection("episodes"))
        }
        deleteCollection(watched)
        deleteCollection(root.collection("shows"))
        deleteCollection(root.collection("members"))
        deleteCollection(root.collection("notes"))
        deleteCollection(root.collection("events"))
        root.delete().await()
        runCatching { messaging.unsubscribeFromTopic("g_$groupId").await() }
    }

    private suspend fun deleteCollection(collection: CollectionReference) {
        while (true) {
            val docs = collection.limit(300).get().await().documents
            if (docs.isEmpty()) return
            db.runBatch { batch ->
                docs.forEach { batch.delete(it.reference) }
            }.await()
        }
    }

    suspend fun addNote(
        groupId: String,
        uid: String,
        authorName: String,
        show: ShowEntry,
        scope: String,
        season: Int,
        episode: Int,
        text: String,
        isPublic: Boolean
    ) {
        val where = when (scope) {
            "season" -> "S${season.toString().padStart(2, '0')} of ${show.name}"
            "episode" -> "S${season.toString().padStart(2, '0')}E${episode.toString().padStart(2, '0')} of ${show.name}"
            else -> show.name
        }
        val groupMemberUids = memberIds(groupId)
        db.runBatch { batch ->
            batch.set(
                db.collection("groups").document(groupId).collection("notes").document(),
                mapOf(
                    "uid" to uid,
                    "authorName" to authorName,
                    "scope" to scope,
                    "showId" to show.id,
                    "season" to season,
                    "episode" to episode,
                    "text" to text,
                    "isPublic" to isPublic,
                    "memberUids" to groupMemberUids,
                    "createdAt" to System.currentTimeMillis()
                )
            )
            if (isPublic) {
                batch.set(
                    db.collection("groups").document(groupId).collection("events").document(),
                    mapOf(
                        "uid" to uid,
                        "authorName" to authorName,
                        "showId" to show.id,
                        "memberUids" to groupMemberUids,
                        "body" to "$authorName wrote a note about $where",
                        "createdAt" to System.currentTimeMillis()
                    )
                )
            }
        }.await()
    }
}

            .background(Brush.linearGradient(listOf(accent.copy(alpha = 0.25f), Brand.Amber.copy(alpha = 0.2f))))
            .border(2.dp, Brand.Stroke, RoundedCornerShape(12.dp)),
        contentAlignment = Alignment.Center
    ) {
        if (url.isNotBlank()) {
            AsyncImage(
                model = url,
                contentDescription = null,

                modifier = Modifier.fillMaxSize()
            )

        } else {

        }
    }
}

@Composable
private fun addedByText(show: ShowEntry, members: List<Member>) {
    Text(
        "Added by ${members.firstOrNull { it.uid == show.addedBy }?.name ?: "someone"}",
        fontSize = 12.sp,
        color = Brand.Muted
    )
}

private fun noteScopeLabel(note: Note) = when (note.scope) {
    "season" -> "Season ${note.season}"
    "episode" -> epLabel(note.season, note.episode)
    else -> "Show note"
}

private fun showName(shows: List<ShowEntry>, showId: String) =
    shows.firstOrNull { it.id == showId }?.name ?: if (showId.isBlank()) "Circle note" else "Removed show"

private fun noteContext(note: Note, shows: List<ShowEntry>) =
    "${showName(shows, note.showId)} · ${noteScopeLabel(note)}"

private fun timeText(ts: Long): String {
    if (ts <= 0L) return ""
    val minutes = (System.currentTimeMillis() - ts) / 60000
    return when {
        minutes < 1 -> "just now"
        minutes < 60 -> "$minutes min ago"
        minutes < 1440 -> "${minutes / 60} h ago"
        else -> "${minutes / 1440} d ago"
    }
}

    var episodes by remember(show.id) { mutableStateOf(cachedEpisodes ?: emptyList()) }
    var loading by remember(show.id) { mutableStateOf(cachedEpisodes == null) }

                ) {

                        val progress = if (episodes.isEmpty()) 0f else done.toFloat() / episodes.size
                        val complete = episodes.isNotEmpty() && done == episodes.size
                        SoftCard(modifier = Modifier.fillMaxWidth()) {
                            Column(Modifier.padding(18.dp)) {
                                Row(verticalAlignment = Alignment.CenterVertically) {
                                    ShowThumb(show.imageUrl, Brand.Blue)
                                    Spacer(Modifier.width(16.dp))
                                    Column(Modifier.weight(1f)) {

                                            show.name.uppercase(),

                                        text = "Progress",
                                        onClick = { go(Screen.GroupProgress(group, show)) },

                                        color = Brand.Blue,
                                        contentColor = Brand.OnPrimary,
                                        leadingIcon = { Icon(Icons.AutoMirrored.Filled.KeyboardArrowRight, contentDescription = null, tint = Brand.OnPrimary) }

                                        contentColor = Brand.OnAccent

package com.dionlobo.watchtogether

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory
import retrofit2.http.GET
import retrofit2.http.Path
import retrofit2.http.Query
import java.util.LinkedHashMap

data class TvImage(val medium: String? = null, val original: String? = null)

data class TvShow(
    val id: Int = 0,
    val name: String = "",
    val image: TvImage? = null,
    val summary: String? = null
)

data class TvSearchResult(val show: TvShow = TvShow())

data class TvEpisodeDto(
    val id: Int = 0,
    val name: String? = null,
    val season: Int? = null,
    val number: Int? = null,
    val airdate: String? = null,
    val image: TvImage? = null
)

interface TvmazeService {
    @GET("search/shows")
    suspend fun search(@Query("q") query: String): List<TvSearchResult>

    @GET("shows/{id}/episodes")
    suspend fun episodes(@Path("id") showId: Int): List<TvEpisodeDto>
}

object TvmazeApi {
    private val service: TvmazeService = Retrofit.Builder()
        .baseUrl("https://api.tvmaze.com/")
        .addConverterFactory(GsonConverterFactory.create())
        .build()
        .create(TvmazeService::class.java)

    private val searchCache = LinkedHashMap<String, List<TvShow>>()
    private val episodeCache = LinkedHashMap<Int, List<TvEpisode>>()

    private fun <K, V> cachePut(cache: LinkedHashMap<K, V>, key: K, value: V, max: Int) {
        synchronized(cache) {
            while (cache.size >= max) cache.remove(cache.keys.first())
            cache[key] = value
        }
    }

    fun cachedSearch(query: String): List<TvShow>? = synchronized(searchCache) {
        searchCache[query.trim().lowercase()]
    }

    suspend fun search(query: String): List<TvShow> = withContext(Dispatchers.IO) {
        val key = query.trim().lowercase()
        synchronized(searchCache) { searchCache[key] }?.let { return@withContext it }
        val result = service.search(query).map { it.show }.filter { it.id != 0 }
        cachePut(searchCache, key, result, 40)
        result
    }

    fun cachedEpisodes(showId: Int): List<TvEpisode>? = synchronized(episodeCache) {
        episodeCache[showId]
    }

    suspend fun episodes(showId: Int): List<TvEpisode> = withContext(Dispatchers.IO) {
        synchronized(episodeCache) { episodeCache[showId] }?.let { return@withContext it }
        val result = service.episodes(showId).map {
            TvEpisode(
                id = it.id,
                name = it.name ?: "Episode ${it.number ?: 0}",
                season = it.season ?: 0,
                number = it.number ?: 0,
                airdate = it.airdate ?: "",
                imageUrl = it.image?.medium ?: ""
            )
        }.sortedWith(compareBy({ it.season }, { it.number }))
        cachePut(episodeCache, showId, result, 24)
        result
    }
}
